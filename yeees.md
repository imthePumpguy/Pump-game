                this.accepted = false;
                this.completed = false;
                this.failed = false;
            }

            getRemainingTime() {
                return Math.max(0, this.deadline - (gameState.tick - this.startTick));
            }

            getProgress() {
                return Math.min(100, (this.wafersDelivered / this.demand) * 100);
            }

            accept() {
                this.accepted = true;
                gameState.log('CONTRACT ACCEPTED: ' + this.name + ' - ' + this.demand + ' wafers by tick ' + (this.startTick + this.deadline), 'success');
            }

            deliver(wafers) {
                if (!this.accepted || this.completed || this.failed) return 0;

                const delivered = Math.min(wafers, this.demand - this.wafersDelivered);
                this.wafersDelivered += delivered;

                if (this.wafersDelivered >= this.demand) {
                    this.completed = true;
                    const payment = this.wafersDelivered * this.payPerWafer;
                    gameState.cash += payment;
                    gameState.totalRevenue += payment;
                    gameState.log('CONTRACT COMPLETED: ' + this.name + ' [$' + payment + ']', 'success');
                }

                return delivered;
            }

            checkDeadline() {
                if (!this.accepted || this.completed || this.failed) return;
                if (this.getRemainingTime() <= 0) {
                    this.failed = true;
                    gameState.log('CONTRACT FAILED: ' + this.name + ' - DEADLINE MISSED', 'error');
                }
            }
        }

        // ============ INITIALIZATION ============
        function initGame() {
            // Add starting pumps
            gameState.pumps['pump1'] = new Pump('qdp80');
            gameState.pumps['pump1'].health = 100;
            gameState.pumps['pump1'].operational = true;

            // Add starting tools
            Object.keys(toolModels).forEach(key => {
                gameState.tools[key] = 0;
            });

            updateUI();
            gameState.log('SYSTEM ONLINE - Facility ready for operation', 'success');
            gameLoop();
        }

        // ============ GAME LOOP ============
        function gameLoop() {
            gameState.tick++;

            // Process contracts
            Object.values(gameState.contracts).forEach(contract => {
                contract.checkDeadline();
            });

            // Generate wafers from pumps
            let wafersProduced = 0;
            Object.values(gameState.pumps).forEach(pump => {
                wafersProduced += pump.tick();
            });

            gameState.wafers += Math.floor(wafersProduced);

            // Pay maintenance
            let maintenanceCost = 0;
            Object.values(gameState.pumps).forEach(pump => {
                maintenanceCost += pump.getMaintenance();
            });

            gameState.totalCosts += maintenanceCost;
            gameState.cash -= maintenanceCost;

            // Deliver to contracts
            let wafersAvailable = gameState.wafers;
            const activeContracts = gameState.contracts.filter(c => c.accepted && !c.completed && !c.failed);

            activeContracts.forEach(contract => {
                const delivered = contract.deliver(wafersAvailable);
                wafersAvailable -= delivered;
                gameState.wafers -= delivered;
            });

            // Bankruptcy check
            if (gameState.cash < 0) {
                gameState.log('BANKRUPTCY - Negative cash flow', 'error');
                gameState.cash = 0;
            }

            // UI update every 10 ticks
            if (gameState.tick % 10 === 0) {
                updateUI();
            }

            requestAnimationFrame(gameLoop);
        }

        // ============ CONTRACT GENERATION ============
        function generateNewContract() {
            const keys = Object.keys(manufacturers);
            const randomMfg = keys[Math.floor(Math.random() * keys.length)];
            const contract = new Contract(randomMfg);
            gameState.contracts.push(contract);
            gameState.log('NEW CONTRACT: ' + contract.name + ' - ' + contract.demand + ' wafers', 'info');
            updateUI();
        }

        // ============ PUMP PURCHASE ============
        function buyPump(modelKey) {
            const model = pumpModels[modelKey];
            if (gameState.cash < model.baseCost) {
                gameState.log('INSUFFICIENT FUNDS for ' + model.name, 'error');
                return;
            }

            const pump = new Pump(modelKey);
            gameState.pumps[pump.id] = pump;
            gameState.cash -= model.baseCost;
            gameState.totalCosts += model.baseCost;
            gameState.log('PUMP PURCHASED: ' + model.name + ' [-$' + model.baseCost + ']', 'success');
            updateUI();
        }

        // ============ TOOL PURCHASE ============
        function buyTool(toolKey) {
            const tool = toolModels[toolKey];
            if (gameState.cash < tool.cost) {
                gameState.log('INSUFFICIENT FUNDS for ' + tool.name, 'error');
                return;
            }

            gameState.tools[toolKey]++;
            gameState.cash -= tool.cost;
            gameState.totalCosts += tool.cost;
            gameState.log('TOOL PURCHASED: ' + tool.name + ' [-$' + tool.cost + ']', 'success');
            updateUI();
        }

        // ============ CONTRACT ACTIONS ============
        function acceptContract(contractId) {
            const contract = gameState.contracts.find(c => c.id === contractId);
            if (contract) {
                contract.accept();
                updateUI();
            }
        }

        // ============ LOGGING ============
        gameState.log = function(message, type = 'info') {
            const logDiv = document.getElementById('systemLog');
            const entry = document.createElement('div');
            entry.className = 'log-entry ' + type;
            entry.textContent = '[T' + gameState.tick + '] ' + message;
            logDiv.insertBefore(entry, logDiv.firstChild);

            while (logDiv.children.length > 20) {
                logDiv.removeChild(logDiv.lastChild);
            }
        };

        // ============ UI UPDATE ============
        function updateUI() {
            // Financial display
            document.getElementById('cashDisplay').textContent = Math.floor(gameState.cash).toLocaleString();
            document.getElementById('wafersDisplay').textContent = Math.floor(gameState.wafers).toLocaleString();
            document.getElementById('revenueDisplay').textContent = Math.floor(gameState.totalRevenue).toLocaleString();
            document.getElementById('costsDisplay').textContent = Math.floor(gameState.totalCosts).toLocaleString();

            // Contracts display
            const contractsContainer = document.getElementById('contractsContainer');
            contractsContainer.innerHTML = '';

            gameState.contracts.forEach(contract => {
                const card = document.createElement('div');
                card.className = 'contract-card';
                if (contract.completed) card.classList.add('completed');
                else if (contract.failed) card.classList.add('failed');
                else if (contract.accepted) card.classList.add('accepted');

                const progress = contract.getProgress();
                const timeLeft = contract.getRemainingTime();

                card.innerHTML = `
                    <div class="contract-header">
                        <div class="contract-logo">
                            <img src="${contract.logo}" alt="${contract.name}" style="filter: brightness(1.5);">
                        </div>
                        <div class="contract-info">
                            <div class="contract-name">${contract.name}</div>
                            <div class="contract-details">
                                ${contract.wafersDelivered}/${contract.demand} wafers
                                <br>
                                ${timeLeft} ticks remaining
                                <br>
                                <span class="contract-payment">$${contract.payPerWafer}/wafer</span>
                            </div>
                        </div>
                    </div>
                    <div class="meter-bar">
                        <div class="meter-fill ${progress < 50 ? 'warning' : ''}" style="width: ${progress}%">
                            ${Math.floor(progress)}%
                        </div>
                    </div>
                    ${!contract.accepted && !contract.completed && !contract.failed ? `
                        <button class="btn success" onclick="acceptContract('${contract.id}')" style="width: 100%; margin-top: 8px;">
                            ACCEPT CONTRACT
                        </button>
                    ` : ''}
                `;

                contractsContainer.appendChild(card);
            });

            // Shop display
            const shopContainer = document.getElementById('shopContainer');
            shopContainer.innerHTML = '';

            Object.entries(pumpModels).forEach(([key, model]) => {
                const item = document.createElement('div');
                item.className = 'shop-item';
                const canAfford = gameState.cash >= model.baseCost;

                item.innerHTML = `
                    <div class="shop-item-header">
                        <span>${model.name}</span>
                        <span class="shop-item-price">$${model.baseCost}</span>
                    </div>
                    <div style="color: #aaa; font-size: 10px; margin-bottom: 6px;">
                        Production: ${model.production}/tick | Failure: ${(model.failureRate * 100).toFixed(1)}%
                    </div>
                    <button class="btn" onclick="buyPump('${key}')" ${!canAfford ? 'disabled' : ''} style="width: 100%;">
                        BUY PUMP
                    </button>
                `;

                shopContainer.appendChild(item);
            });

            // Tools display
            const toolsContainer = document.getElementById('toolsContainer');
            toolsContainer.innerHTML = '';

            Object.entries(toolModels).forEach(([key, tool]) => {
                const item = document.createElement('div');
                item.className = 'shop-item';
                const canAfford = gameState.cash >= tool.cost;
                const quantity = gameState.tools[key] || 0;

                item.innerHTML = `
                    <div class="shop-item-header">
                        <span>${tool.name}</span>
                        <span class="shop-item-price">$${tool.cost}</span>
                    </div>
                    <div style="color: #aaa; font-size: 10px; margin-bottom: 6px;">
                        ${tool.effect} | Owned: ${quantity}
                    </div>
                    <button class="btn" onclick="buyTool('${key}')" ${!canAfford ? 'disabled' : ''} style="width: 100%;">
                        BUY
                    </button>
                `;

                toolsContainer.appendChild(item);
            });

            // Pumps display
            const pumpsContainer = document.getElementById('pumpsContainer');
            pumpsContainer.innerHTML = '';

            Object.values(gameState.pumps).forEach(pump => {
                const card = document.createElement('div');
                card.className = 'equipment-card';

                const statusText = pump.operational ? 'OPERATIONAL' : 'FAILED';
                const statusColor = pump.operational ? '#00ff00' : '#ff0000';

                card.innerHTML = `
                    <div class="equipment-name">${pump.name}</div>
                    <div class="equipment-status">Status: <span style="color: ${statusColor}; font-weight: bold;">${statusText}</span></div>
                    <div class="meter-bar">
                        <div class="meter-fill ${pump.health < 50 ? 'warning' : ''}" style="width: ${pump.health}%; background: linear-gradient(90deg, #00ff00, #ffaa00);">
                            ${Math.floor(pump.health)}%
                        </div>
                    </div>
                    <div style="color: #aaa; font-size: 10px; margin-top: 6px;">
                        Runtime: ${pump.runtime} | Maintenance: $${Math.round(pump.getMaintenance())}
                    </div>
                `;

                pumpsContainer.appendChild(card);
            });
        }

        // Start the game
        window.addEventListener('load', initGame);
    </script>
</body>
</html>
